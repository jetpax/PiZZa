# config.txt template for booting Zephyr on a Raspberry Pi Zero 2 W
# from the SD card's FAT boot partition (see make-sdcard.sh).
#
# install-to-sdcard.sh writes this content automatically; the file
# is committed here so reviewers can see the boot config without
# reading shell code.

arm_64bit=1
# enable_uart=1 routes the mini-UART to GPIO 14/15 and locks the core
# clock so its baud divisor stays valid.
enable_uart=1
core_freq=250
kernel_address=0x200000
kernel=zephyr.bin

# Bring HDMI up at boot even if no monitor is attached when the Pi is
# powered on. Without this, hot-plugging a cable later leaves the
# pipeline cold and the Zephyr FB driver's ALLOCATE_BUFFER returns
# an empty buffer. Harmless when HDMI isn't used.
hdmi_force_hotplug=1

# Force HDMI mode (not DVI): DVI carries no audio packets, and with
# hdmi_force_hotplug the firmware otherwise falls back to DVI when
# EDID is absent or ambiguous. Required for HDMI audio.
hdmi_drive=2

# Hand the whole panel to the framebuffer. The firmware's default
# overscan compensation insets 48 px on every side, so a 1080p mode
# allocates 1824x984 and an app that trusts display_get_capabilities
# draws a picture that does not reach the edges. Nothing on a modern
# monitor needs the inset.
disable_overscan=1
