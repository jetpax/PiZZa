# config.txt template for booting Zephyr on an original Raspberry Pi
# Zero W (BCM2835, ARM1176JZF-S, ARMv6KZ) -- 32-bit kernel.
# Used by make-sdcard.sh for rpi_zero_w images.

# BCM2835 / ARM1176 is 32-bit; GPU loads the raw kernel at 0x8000.
kernel=zephyr.bin
kernel_address=0x8000

# This port does not consume ATAGS; tell the firmware not to write them.
disable_commandline_tags=1

# Console is the AUX mini-UART (UART1) on GPIO 14/15. enable_uart=1 makes
# the firmware enable the AUX block, mux GPIO 14/15 to the mini-UART, and
# pin the core clock so the mini-UART baud divisor stays valid (its clock
# is the core clock, unlike PL011). PL011 is left on the Bluetooth modem.
enable_uart=1
core_freq=250

# Bring HDMI up at boot even with no monitor attached at power-on.
# Without this, hot-plugging later leaves the pipeline cold and the
# framebuffer driver's ALLOCATE_BUFFER returns an empty buffer.
# Harmless when HDMI isn't used.
hdmi_force_hotplug=1

# Hand the whole panel to the framebuffer; the firmware's default
# overscan compensation insets 48 px on every side.
disable_overscan=1
