# Minimal config.txt for Pi Zero 2W with rpi-open-firmware
#
# rpi-open-firmware replaces the stock bootcode.bin, so most
# standard config.txt options don't apply. This file is read
# by the GPU ROM before our firmware runs.
#
# See: https://www.raspberrypi.com/documentation/computers/config_txt.html

# Force Pi Zero 2W mode (BCM2710)
# Uncomment if auto-detection fails:
# force_board_rev=0xc03111

# Disable unnecessary hardware to save power
dtoverlay=disable-bt
dtoverlay=disable-wifi

# UART output for debugging (GPIO14=TXD, GPIO15=RXD)
enable_uart=1

# GPU memory split — give minimum to GPU since we're bare-metal
gpu_mem=16

# Don't try to load start.elf (we ARE the firmware)
# This should be handled by bootcode.bin, but just in case:
# kernel=none

# Clock settings — conservative for stability during experiments
# arm_freq=600
# core_freq=250
